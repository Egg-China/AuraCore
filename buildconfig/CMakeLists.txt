string(TIMESTAMP AURACORE_BUILD_TIMESTAMP "%Y-%m-%dT%H:%M:%SZ" UTC)
execute_process(COMMAND git rev-parse HEAD
    OUTPUT_VARIABLE AURACORE_GIT_COMMIT
    OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_QUIET
    RESULT_VARIABLE AURACORE_GIT_COMMIT_RESULT)
execute_process(COMMAND git describe --exact-match --tags
    OUTPUT_VARIABLE AURACORE_GIT_TAG
    OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_QUIET
    RESULT_VARIABLE AURACORE_GIT_TAG_RESULT)
execute_process(COMMAND git symbolic-ref -q --short HEAD
    OUTPUT_VARIABLE AURACORE_GIT_REFSPEC
    OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_QUIET
    RESULT_VARIABLE AURACORE_GIT_REFSPEC_RESULT)

if(AURACORE_GIT_COMMIT_RESULT EQUAL 0)
    set(AURACORE_GIT_COMMIT "${AURACORE_GIT_COMMIT}")
else()
    set(AURACORE_GIT_COMMIT "")
endif()
if(AURACORE_GIT_TAG_RESULT EQUAL 0)
    set(AURACORE_GIT_TAG "${AURACORE_GIT_TAG}")
else()
    set(AURACORE_GIT_TAG "")
endif()
if(AURACORE_GIT_REFSPEC_RESULT EQUAL 0)
    set(AURACORE_GIT_REFSPEC "refs/heads/${AURACORE_GIT_REFSPEC}")
else()
    set(AURACORE_GIT_REFSPEC "")
endif()

add_library(BuildConfig STATIC
    BuildConfig.h
    BuildConfig.cpp
)
target_include_directories(BuildConfig PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(BuildConfig PRIVATE
    AURACORE_VERSION_MAJOR=${PROJECT_VERSION_MAJOR}
    AURACORE_VERSION_MINOR=${PROJECT_VERSION_MINOR}
    AURACORE_VERSION_PATCH=${PROJECT_VERSION_PATCH}
    AURACORE_BUILD_PLATFORM="${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}"
    AURACORE_BUILD_TIMESTAMP="${AURACORE_BUILD_TIMESTAMP}"
    AURACORE_COMPILER_NAME="${CMAKE_CXX_COMPILER_ID}"
    AURACORE_COMPILER_VERSION="${CMAKE_CXX_COMPILER_VERSION}"
    AURACORE_GIT_COMMIT="${AURACORE_GIT_COMMIT}"
    AURACORE_GIT_TAG="${AURACORE_GIT_TAG}"
    AURACORE_GIT_REFSPEC="${AURACORE_GIT_REFSPEC}"
)
target_link_libraries(BuildConfig PUBLIC Qt6::Core)
